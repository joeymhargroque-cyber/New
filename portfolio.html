import java.util.Scanner;

public class Arrange the grade {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        double[] grades = new double[15];

        double sum = 0;
        double average;

        int above = 0;
        int below = 0;
        int equal = 0;

        // Input 15 grades
        for (int i = 0; i < grades.length; i++) {

            while (true) {
                System.out.print("Enter grade of student " + (i + 1) + " (0-100): ");

                double grade = input.nextDouble();

                if (grade >= 0 && grade <= 100) {
                    grades[i] = grade;
                    sum += grades[i];
                    break;
                } else {
                    System.out.println("Invalid grade! Enter 0 to 100 only.");
                }
            }
        }

        // Calculate average
        average = sum / grades.length;

        // Find highest and lowest grades
        double highest = grades[0];
        double lowest = grades[0];

        for (int i = 0; i < grades.length; i++) {

            if (grades[i] > highest) {
                highest = grades[i];
            }

            if (grades[i] < lowest) {
                lowest = grades[i];
            }

            // Compare grades with average
            if (grades[i] > average) {
                above++;
            } else if (grades[i] < average) {
                below++;
            } else {
                equal++;
            }
        }

        // Manual ascending sorting (Bubble Sort)
        for (int i = 0; i < grades.length - 1; i++) {

            for (int j = 0; j < grades.length - 1 - i; j++) {

                if (grades[j] > grades[j + 1]) {

                    double temp = grades[j];
                    grades[j] = grades[j + 1];
                    grades[j + 1] = temp;

                }
            }
        }

        // Display results
        System.out.println("\n===== MIDTERM GRADE RESULTS =====");

        System.out.printf("Average Grade: %.2f%n", average);
        System.out.println("Grades Above Average: " + above);
        System.out.println("Grades Below Average: " + below);
        System.out.println("Grades Equal to Average: " + equal);

        System.out.println("Highest Grade: " + highest);
        System.out.println("Lowest Grade: " + lowest);

        System.out.println("\nGrades in Ascending Order:");

        for (int i = 0; i < grades.length; i++) {
            System.out.println("Index " + i + ": " + grades[i]);
        }

        input.close();
    }
}
